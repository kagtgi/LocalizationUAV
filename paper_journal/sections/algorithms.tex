\begin{algorithm}[t]
\caption{Offline map preparation (training-free)}
\label{alg:offline}
\begin{algorithmic}[1]
\Require orthophoto $\mathcal S$ (GSD $g_S$), working GSD $g$, bins $K$, min.\ length $\ell_{\min}$
\State segments $\mathcal L_S\gets\mathrm{LSD}(\mathcal S)$, keep length $\ge\ell_{\min}$
\State rasterize $\mathcal L_S$ at $g$ with orientation labels $b\in\{1..K\}$
\For{$k=1..K$} $\kappa_k\gets\exp(-D_k^2/2\sigma^2)$ \EndFor
\State $G_k\gets\max(\kappa_k,\tfrac12\kappa_{k\pm1})$ \Comment{soft orientation tolerance}
\State line graph: filter/merge $\mathcal L_S$, junctions $\{\mathbf J_j\}$, adjacency, Ekeland angles $e_j$; $k$-d tree
\State \Return $\{G_k\}$, line graph
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Online localization of one UAV frame}
\label{alg:online}
\begin{algorithmic}[1]
\Require image $I$, altitude $h$, heading $\psi$, prior $\bar{\mathbf p}$, radius $R$
\State $I'\gets$ rotate by $-\psi$, resample to $g_0$ with $g_I=kh$; footprint $V$
\State oriented query lines; ensemble $\mathcal E$ $\Rightarrow$ persistence $p_b$ \eqref{eq:persistence}; samples $(\mathbf x_b,\phi_b,w_b)$
\For{each $(\theta,s)\in\Theta\times\mathcal S$} \Comment{batched on GPU, templates built lazily}
  \State per-channel templates $\mathcal T_k$ (samples with $\mathrm{bin}(\phi_b+\theta)=k$), footprint template
  \State $J_{\theta,s}(\mathbf t)\gets$ masked stacked NCC via FFT (Prop.~\ref{prop:fft}) $-\lambda\,\mathrm{prior}(\theta,s)$
  \State $J^\ast(\mathbf t)\gets\max(J^\ast(\mathbf t),J_{\theta,s}(\mathbf t))$ within the window
\EndFor
\State top-$k$ peaks (NMS 30\,m); runner-up $J_2$
\State query line graph; re-rank peaks by $J+\gamma A(T)$ \Comment{junctions, topology, Ekeland}
\State continuous maximization of $J$; RANSAC Sim(2) on verified junctions
\State $\hat\Sigma_{\mathbf p}\gets$ Laplace \eqref{eq:laplace}; $c\gets\sigma(\boldsymbol\beta^\top\mathbf z)$
\If{$c<\tau_c$} \Return abstain \EndIf
\State \Return $\hat{\mathbf p}$, $\hat\Sigma_{\mathbf p}$, $c$
\end{algorithmic}
\end{algorithm}
